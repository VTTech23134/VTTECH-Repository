package org.firstinspires.ftc.teamcode;

import com.qualcomm.robotcore.eventloop.opmode.LinearOpMode;
import com.qualcomm.robotcore.eventloop.opmode.TeleOp;
import com.qualcomm.robotcore.hardware.DcMotor;

@TeleOp(name = "Test: frontRightMotor", group = "Test")
public class TestFrontRightMotor extends LinearOpMode {

    private DcMotor frontRightMotor;

    @Override
    public void runOpMode() {
        // Verifica che il nome corrisponda esattamente alla configurazione del Robot Controller
        frontRightMotor = hardwareMap.get(DcMotor.class, "frontRightMotor");

        frontRightMotor.setMode(DcMotor.RunMode.STOP_AND_RESET_ENCODER);
        frontRightMotor.setMode(DcMotor.RunMode.RUN_USING_ENCODER);
        frontRightMotor.setZeroPowerBehavior(DcMotor.ZeroPowerBehavior.BRAKE);
        frontRightMotor.setPower(0.0);

        telemetry.addData("Stato", "Pronto per il test. Premi Play.");
        telemetry.update();

        waitForStart();

        double potenza = 0.0;
        double incremento = 0.01;
        boolean inAumento = true;

        while (opModeIsActive()) {
            if (inAumento) {
                potenza = Math.min(1.0, potenza + incremento);
                if (potenza >= 1.0) {
                    inAumento = false;
                }
            } else {
                potenza = Math.max(-1.0, potenza - incremento);
                if (potenza <= -1.0) {
                    inAumento = true;
                }
            }

            frontRightMotor.setPower(potenza);

            telemetry.addData("Potenza attuale", "%.2f", potenza);
            telemetry.addData("Posizione Encoder (tick)", frontRightMotor.getCurrentPosition());
            telemetry.update();

            sleep(30);
        }

        frontRightMotor.setPower(0.0);
    }
}
